## 🗄️ DATABASE SCHEMA

### Core Tables

```sql
-- Tours
CREATE TABLE tours (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES auth.users NOT NULL,
  organization_id UUID REFERENCES organizations,
  name TEXT NOT NULL,
  artist_name TEXT NOT NULL,
  status TEXT CHECK (status IN ('planning', 'active', 'completed', 'cancelled')),
  start_date DATE,
  end_date DATE,
  description TEXT,
  budget DECIMAL(12,2),
  currency TEXT DEFAULT 'USD',
  tour_type TEXT, -- 'headline', 'support', 'festival', 'club'
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tour Dates
CREATE TABLE tour_dates (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_id UUID REFERENCES tours ON DELETE CASCADE,
  date DATE NOT NULL,
  venue_id UUID REFERENCES venues,
  city TEXT,
  state_province TEXT,
  country TEXT,
  status TEXT CHECK (status IN ('confirmed', 'hold', 'pending', 'cancelled')),
  show_time TIME,
  doors_time TIME,
  soundcheck_time TIME,
  load_in_time TIME,
  curfew_time TIME,
  ticket_price DECIMAL(10,2),
  ticket_url TEXT,
  capacity INTEGER,
  expected_attendance INTEGER,
  actual_attendance INTEGER,
  revenue DECIMAL(12,2),
  expenses DECIMAL(12,2),
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Venues
CREATE TABLE venues (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  address TEXT,
  city TEXT,
  state_province TEXT,
  country TEXT,
  postal_code TEXT,
  latitude DECIMAL(10,8),
  longitude DECIMAL(11,8),
  phone TEXT,
  email TEXT,
  website TEXT,
  capacity INTEGER,
  venue_type TEXT, -- 'arena', 'theater', 'club', 'festival', 'outdoor'
  stage_size TEXT,
  load_in_notes TEXT,
  parking_notes TEXT,
  amenities JSONB,
  technical_specs JSONB,
  contacts JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Itinerary Items
CREATE TABLE itinerary_items (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  item_type TEXT CHECK (item_type IN ('show', 'travel', 'hotel', 'rehearsal', 'meeting', 'promo', 'day_off', 'load_in', 'soundcheck', 'other')),
  title TEXT NOT NULL,
  start_time TIMESTAMPTZ,
  end_time TIMESTAMPTZ,
  location TEXT,
  description TEXT,
  participants JSONB, -- array of crew/personnel IDs
  status TEXT DEFAULT 'scheduled',
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Crew/Personnel
CREATE TABLE tour_crew (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_id UUID REFERENCES tours ON DELETE CASCADE,
  user_id UUID REFERENCES auth.users,
  name TEXT NOT NULL,
  role TEXT NOT NULL,
  email TEXT,
  phone TEXT,
  emergency_contact JSONB,
  dietary_restrictions TEXT,
  passport_info JSONB,
  travel_preferences JSONB,
  permissions TEXT CHECK (permissions IN ('admin', 'manager', 'crew', 'view_only')),
  visibility TEXT[] DEFAULT ARRAY['all'], -- which crews can see this person
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Hotels
CREATE TABLE hotels (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  name TEXT NOT NULL,
  address TEXT,
  city TEXT,
  state_province TEXT,
  country TEXT,
  postal_code TEXT,
  phone TEXT,
  email TEXT,
  website TEXT,
  check_in DATE,
  check_out DATE,
  confirmation_number TEXT,
  rate DECIMAL(10,2),
  notes TEXT,
  amenities JSONB,
  contacts JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Hotel Room Lists
CREATE TABLE hotel_rooms (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  hotel_id UUID REFERENCES hotels ON DELETE CASCADE,
  crew_member_id UUID REFERENCES tour_crew,
  room_number TEXT,
  room_type TEXT,
  smoking BOOLEAN DEFAULT false,
  bag_tag TEXT,
  confirmation_number TEXT,
  special_requests TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Travel
CREATE TABLE travel_items (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates,
  travel_type TEXT CHECK (travel_type IN ('air', 'ground', 'rail', 'sea')),
  departure_location TEXT,
  arrival_location TEXT,
  departure_time TIMESTAMPTZ,
  arrival_time TIMESTAMPTZ,
  airline TEXT,
  flight_number TEXT,
  flightaware_id TEXT, -- for real-time tracking
  confirmation_number TEXT,
  transport_company TEXT,
  vehicle_type TEXT,
  passengers JSONB, -- array of crew member IDs
  cost DECIMAL(10,2),
  notes TEXT,
  tracking_status TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Guest Lists
CREATE TABLE guest_lists (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  requester_id UUID REFERENCES auth.users,
  guest_name TEXT NOT NULL,
  guest_email TEXT,
  guest_phone TEXT,
  plus_ones INTEGER DEFAULT 0,
  pass_type TEXT, -- 'VIP', 'Guest', 'Comp', 'Photo', 'Working', etc.
  status TEXT CHECK (status IN ('pending', 'approved', 'declined')) DEFAULT 'pending',
  approved_by UUID REFERENCES auth.users,
  approved_at TIMESTAMPTZ,
  notes TEXT,
  checked_in BOOLEAN DEFAULT false,
  checked_in_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Guest List Allotments
CREATE TABLE guest_list_allotments (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  pass_type TEXT NOT NULL,
  total_allotment INTEGER NOT NULL,
  enforce BOOLEAN DEFAULT true,
  cutoff_time TIMESTAMPTZ,
  locked BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Set Lists
CREATE TABLE songs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES auth.users,
  title TEXT NOT NULL,
  artist TEXT,
  duration INTEGER, -- seconds
  bpm INTEGER,
  key TEXT,
  tech_notes TEXT,
  attachments JSONB, -- array of file URLs
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE setlists (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  name TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE setlist_songs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  setlist_id UUID REFERENCES setlists ON DELETE CASCADE,
  song_id UUID REFERENCES songs,
  position INTEGER NOT NULL,
  is_break BOOLEAN DEFAULT false,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Financial Tracking
CREATE TABLE tour_expenses (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_id UUID REFERENCES tours,
  tour_date_id UUID REFERENCES tour_dates,
  category TEXT, -- 'travel', 'hotel', 'food', 'fuel', 'equipment', 'misc'
  amount DECIMAL(10,2) NOT NULL,
  currency TEXT DEFAULT 'USD',
  description TEXT,
  receipt_url TEXT,
  submitted_by UUID REFERENCES auth.users,
  approved_by UUID REFERENCES auth.users,
  approved_at TIMESTAMPTZ,
  status TEXT CHECK (status IN ('pending', 'approved', 'rejected')) DEFAULT 'pending',
  date DATE NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE tour_revenue (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  source TEXT, -- 'tickets', 'merch', 'meet_greet', 'guarantee', 'other'
  amount DECIMAL(12,2) NOT NULL,
  currency TEXT DEFAULT 'USD',
  description TEXT,
  received_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- AI Predictions & Analytics
CREATE TABLE tour_analytics (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tour_date_id UUID REFERENCES tour_dates ON DELETE CASCADE,
  predicted_attendance INTEGER,
  predicted_revenue DECIMAL(12,2),
  predicted_expenses DECIMAL(12,2),
  risk_score DECIMAL(5,2), -- 0-100
  sentiment_score DECIMAL(5,2), -- -1 to 1
  recommendations JSONB,
  generated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Notifications
CREATE TABLE notifications (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES auth.users NOT NULL,
  tour_id UUID REFERENCES tours,
  notification_type TEXT NOT NULL,
  title TEXT NOT NULL,
  message TEXT,
  data JSONB,
  read BOOLEAN DEFAULT false,
  read_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Organizations (for multi-user management)
CREATE TABLE organizations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  owner_id UUID REFERENCES auth.users,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE organization_members (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID REFERENCES organizations ON DELETE CASCADE,
  user_id UUID REFERENCES auth.users ON DELETE CASCADE,
  role TEXT CHECK (role IN ('owner', 'admin', 'manager', 'member', 'view_only')),
  permissions JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(organization_id, user_id)
);
```

## 🔐 SECURITY & PERMISSIONS

### Row Level Security (RLS)
```sql
-- Users can only see their own tours
CREATE POLICY "Users can view own tours"
  ON tours FOR SELECT
  USING (auth.uid() = user_id OR auth.uid() IN (
    SELECT user_id FROM organization_members
    WHERE organization_id = tours.organization_id
  ));

-- Granular permissions for crew
CREATE POLICY "Crew visibility based on permissions"
  ON tour_crew FOR SELECT
  USING (
    auth.uid() IN (
      SELECT user_id FROM tour_crew
      WHERE tour_id = tour_crew.tour_id
      AND permissions IN ('admin', 'manager')
    )
  );
```
